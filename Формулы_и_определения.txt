ТСП - толщина сухой пленки, мкм
ТМП - толщина мокрой пленки, мкм
%СО - объемная доля сухого остатка, %
%Раствор - процент разбавления растворителем, %
КП - коэффициент потерь
КП=100%/(100%-потери(%))
ТСП=(ТМП*%СО)/100, мкм
ТМП=(ТСП*100)/%СО, мкм
ТМП=(ТСП*(100+%раствор))/%СО, мкм
Кол-во краски (теор.расход) = (1(м^2)*ТСП)/(10*%СО), л/м^2
Кол-во краски (практ.расход) = (1(м^2)*ТСП)/(10*%СО*(1-КП)), л/м^2
Кол-во краски (теор.расход) = ((1(м^2)*ТСП)/(10*%СО))*плотность, кг/м^2
Кол-во краски (практ.расход) = (1(м^2)*ТСП)/(10*%СО*(1-КП))*плотность, кг/м^2
